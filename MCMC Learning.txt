<?php
// PREVENT UNAUTHORIZED ACCESS ASAP

// Load Joomla (keep JFactory for your stack)
if (!defined('JPATH_BASE')) {
    define('JPATH_BASE', realpath(dirname(__FILE__) . '/../..'));

if (!defined('_JEXEC')) {

// -----------------------------------------------------------------------------
// HELPERS (must be defined before first use)

if (!function_exists('quoteTable')) {
    // Safely quote a table name with Joomla prefix replacement
    function quoteTable($db, $name) {
        $real = str_replace('#__', $db->getPrefix(), $name);
        return $db->quoteName($real);
    }
}

if (!function_exists('resolveGameIdDb')) {
    /**
     * Resolve a canonical DB game_id given a request/JSON key (case-insensitive).
     * Returns the exact game_id string as stored in #__lotteries, or the original
     * if not found (for backward compatibility with legacy tables).
     */
    function resolveGameIdDb($db, $requestedGameId) {
        if (!$requestedGameId) return '';
        $q = $db->getQuery(true)
            ->select($db->quoteName('game_id'))
            ->from($db->quoteName('#__lotteries'))
            ->where('LOWER(' . $db->quoteName('game_id') . ') = LOWER(' . $db->quote($requestedGameId) . ')')
            ->setLimit(1);
        $db->setQuery($q);
        $res = $db->loadResult();
        return $res ?: $requestedGameId; // graceful fallback
    }
}

if (!function_exists('getNextDrawDate')) {
    /**
     * Fetch the next_draw_date for this game from its OWN draw table (from dbCol),
     * not from #__lotteries. We select MAX(next_draw_date) for the latest scheduled draw.
     */
    function getNextDrawDate($gameIdDb, $db, $tableName) {
        // Use the game-specific table defined in lottery_skip_config.json (dbCol)
        $q = $db->getQuery(true)
            ->select('MAX(' . $db->quoteName('next_draw_date') . ')')
            ->from(quoteTable($db, $tableName))
            ->where($db->quoteName('game_id') . ' = ' . $db->quote($gameIdDb));

        $db->setQuery($q);
        $res = $db->loadResult();

        // Return a safe string
        return $res ? (string) $res : '';
    }
}

$input = JFactory::getApplication()->input;

// Try multiple param names first (query/body/SEF-friendly)
$rawGameId = $input->getString('gameId')
           ?: $input->getString('game_id')
           ?: ($_GET['gameId'] ?? $_GET['game_id'] ?? null);

// If still empty, sniff from the request URI
if (!$rawGameId && !empty($_SERVER['REQUEST_URI'])) {
    $uriPath = parse_url($_SERVER['REQUEST_URI'], PHP_URL_PATH) ?: '';
    $uriPath = rtrim($uriPath, '/');

    // accept ?game_id=... or ?gameId=...
    if (preg_match('~[?&](game_id|gameId)=([A-Za-z0-9_-]+)~', $_SERVER['REQUEST_URI'], $m)) {
        $rawGameId = $m[2];
    }
    // last path segment could be the id (allow longer than 4)
    elseif (preg_match('~/([A-Za-z0-9_-]{2,})$~', $uriPath, $m)) {
        $rawGameId = $m[1];
    }
}

// DO NOT force upper only; try exact, then upper, then lower
$requestedId = trim((string)$rawGameId);

// Include Joomla authentication early (kept from your version)
include '../joomla-auth.php';

// Auth gate
$currUser       = JFactory::getUser();

$userIsInAllowedGroup = ($currUser->guest && in_array(1, $allowedGroups))

    : !empty(array_intersect($allowedGroups, $userGroups));

// -----------------------------------------------------------------------------

$jsonFile = '/home/oscara/web/lottoexpert.net/public_html/lottery_skip_config.json';

$lotteries = isset($cfgAll['lotteries']) && is_array($cfgAll['lotteries']) ? $cfgAll['lotteries'] : [];

// Resolve the gameId case-insensitively

    $gameId = strtoupper($requestedId);
} elseif ($requestedId !== '' && isset($lotteries[strtolower($requestedId)])) {
    $gameId = strtolower($requestedId);

// Defaulting logic
$DEFAULT_GAME_ID = '101';
if ($gameId === '') {

        ->select('l.game_id')
        ->from( $db->quoteName('#__lotteries','l') )
        ->where( $db->quoteName('l.lottery_id') . ' = ' . $db->quote($lottery) );

    $gameDbId = $db->loadResult();

    $nextDraw = getNextDrawDate($gameDbId, $db, $tableName);

    // Insert
    $ins = $db->getQuery(true)
        ->insert($db->quoteName('#__user_saved_numbers'))
        ->columns([
            'user_id','lottery_id','main_numbers','extra_ball_numbers',
            'source','label','generated_at','date_saved','next_draw_date',

            . ", " . $saved
            . ", " . $db->quote($nextDraw)
            . ", " . $W
            . ", " . $B
            . ", " . $K
            . ", " . $D
            . ", " . $CL

        $db->setQuery($ins);

        $err = $db->getErrorMsg();

    ->where($db->quoteName('game_id') . ' = ' . $db->quote($gameIdDb))

foreach ($rows as $r) {

    foreach ($cols as $c) {
        $val = isset($r[$c]) ? (int)$r[$c] : 0;
        if ($val === 0 && !$isDailyGame) continue; // drop zero for non daily
        $seq[] = $val;
    }
    if (!empty($seq)) {
        $draws[] = $seq;
        // [ADD BT] emit a normalized Y-m-d date string if present
        $dt = isset($r['draw_date']) ? (string)$r['draw_date'] : '';
        $drawDates[] = $dt !== '' ? date('Y-m-d', strtotime($dt)) : '';
    }

    var btContainer = document.getElementById('btCard');

    // Convert [[...]] ? <...> and strip any non-whitelisted tags for safety
    function decodeToHtml(html) {
      if (!html || html.indexOf('[[') === -1) return html;
      // Replace bracket tokens
      var out = html.replace(/\[\[/g, '<').replace(/]]/g, '>');
      // Whitelist only a minimal set of tags used in the BT UI
      // (add more if your BT renderer legitimately uses them)
      out = out.replace(
        /<(?!/?(?:span|div|br|table|thead|tbody|tr|th|td|ul|li|em|strong|code|small|b|i)\b)[^>]*>/gi,
        function (m) { return m.replace(/</g,'&lt;').replace(/>/g,'&gt;'); }
      );
      return out;
    }

    // Walk all descendant elements and only fix nodes that actually contain [[...]]
    function fixBrackets(root) {
      // elements likely updated by the BT renderer
      var scopeNodes = root.querySelectorAll('*');
      for (var i = 0; i < scopeNodes.length; i++) {
        var el = scopeNodes[i];
        // Skip inputs/buttons to avoid breaking listeners
        if (el.tagName === 'INPUT' || el.tagName === 'BUTTON' || el.tagName === 'SELECT' || el.tagName === 'OPTION') continue;
        var html = el.innerHTML;
        if (html && html.indexOf('[[') !== -1) {
          el.innerHTML = decodeToHtml(html);
        }
      }
    }

    // Initial pass (in case BT content is pre-rendered)
    fixBrackets(btContainer);

    // Observe future updates (when the backtest populates rows/cells)
    var mo = new MutationObserver(function (muts) {
      // Only touch nodes that changed under btContainer
      fixBrackets(btContainer);
    });
    mo.observe(btContainer, { childList: true, subtree: true });
  });

    Each of the five settings below controls how the Markov Chain Monte Carlo algorithm explores past lottery draws and produces predictions.
    Adjusting them changes both the speed of computation and the stability (variance vs. bias) of results.
    Use the “Fast” configuration for quick tests, and the “Thorough” configuration when you need the most reliable output.
  [[/p]]

  [[h4 style="color:#2a5298; margin-top:20px;"]]1. Simulations (Number of MCMC Walks)[[/h4]]
  [[p style="font-size:14px; color:#444; margin-bottom:10px;"]]
    [[strong]]Range:[[/strong]] 100 to 40&nbsp;000[[br]]
    [[strong]]Effect:[[/strong]] Number of independent random walks.
    More walks &rarr; more samples &rarr; lower variance in the final visit-frequency estimates; fewer walks &rarr; faster but noisier.
  [[/p]]

  [[h4 style="color:#2a5298; margin-top:20px;"]]2. Burn-In (Discard Initial Steps)[[/h4]]
  [[p style="font-size:14px; color:#444; margin-bottom:10px;"]]
    [[strong]]Range:[[/strong]] 0 to 2&nbsp;000[[br]]
    [[strong]]Effect:[[/strong]] Discards the first steps of each walk so scoring focuses on the steady-state portion.
    Larger burn-in &rarr; safer mixing; smaller &rarr; faster but may keep transient bias.
  [[/p]]

  [[h4 style="color:#2a5298; margin-top:20px;"]]3. Laplace K (Smoothing Constant)[[/h4]]
  [[p style="font-size:14px; color:#444; margin-bottom:10px;"]]
    [[strong]]Range:[[/strong]] 0.1 (or 0) to 10+[[br]]
    [[strong]]Effect:[[/strong]] Adds a small count to every transition to avoid zero probabilities.
    Higher K &rarr; heavier smoothing (flatter differences); lower K &rarr; sharper peaks but risk of overfitting.
  [[/p]]

  [[h4 style="color:#2a5298; margin-top:20px;"]]4. Recency Decay (Weight Recent Draws More)[[/h4]]
  [[p style="font-size:14px; color:#444; margin-bottom:10px;"]]
    [[strong]]Range:[[/strong]] 0.001 to 1.0[[br]]
    [[strong]]Effect:[[/strong]] Controls how quickly older draws lose influence.
    Higher decay &rarr; strong emphasis on the newest draws; lower decay &rarr; more uniform influence over history.
  [[/p]]

  [[h4 style="color:#2a5298; margin-top:20px;"]]5. Chain Length (Steps per Walk)[[/h4]]
  [[p style="font-size:14px; color:#444; margin-bottom:15px;"]]
    [[strong]]Range:[[/strong]] 100 to 10&nbsp;000[[br]]
    [[strong]]Effect:[[/strong]] Steps taken per walk after burn-in.
    Longer chains &rarr; deeper exploration and better mixing; shorter &rarr; faster but shallower.
  [[/p]]

function formatEtaDHMS(seconds){
  if(!isFinite(seconds) || seconds < 0) return "—";
  var s = Math.floor(seconds);
  var d = Math.floor(s / 86400); s %= 86400;
  var h = Math.floor(s / 3600);  s %= 3600;
  var m = Math.floor(s / 60);    s %= 60;
  var parts = [];
  if (d > 0) parts.push(d + "d");
  if (h > 0) parts.push(h + "h");
  if (m > 0) parts.push(m + "m");
  parts.push(s + "s");
  return parts.join(" ");
}

  // Build evaluation checkpoints with slight recency emphasis (geometric spacing).
  // start = t < endExclusive-1 (since we use truth = draws[t+1])
  function buildCheckpoints(start, endExclusive, casesMin, casesMax){
    var span = endExclusive - start;
    if (span <= 0) return [];
    var target = Math.round(Math.sqrt(span)) * 2;               // base on vspan
    var cases  = Math.max(casesMin, Math.min(casesMax, target)); // clamp for stability
    var out = [];
    for (var i = 1; i <= cases; i++){
      var q = i / (cases + 1);
      var s = Math.pow(q, 1.4);                 // 1.0 = uniform, <2 favors recency a bit
      var t = start + Math.floor(s * span);
      if (t >= endExclusive - 1) t = endExclusive - 2;
      if (out.length === 0 || out[out.length - 1] !== t) out.push(t);
    }
    return out;
  }

  function stationaryDistribution(rowStochastic, maxIter, tol){
    var n = rowStochastic.length;
    if (!n) return [];
    var p = new Float64Array(n);
    var one = 1 / n;
    for (var i = 0; i < n; i++) p[i] = one;
    maxIter = maxIter || 256;
    tol = tol || 1e-10;
    var pNext = new Float64Array(n);

    for (var it = 0; it < maxIter; it++){
      for (var j = 0; j < n; j++) pNext[j] = 0;
      for (var r = 0; r < n; r++){
        var pr = p[r];
        var row = rowStochastic[r];
        for (var c = 0; c < n; c++) { pNext[c] += pr * row[c]; }
      }
      var diff = 0;
      for (var k = 0; k < n; k++){
        var d = Math.abs(pNext[k] - p[k]);
        diff += d;
        p[k] = pNext[k];
      }
      if (diff < tol) break;
    }
    return Array.from(p);
  }

  function blendProbs(arrays, weights){
    if (!arrays || !arrays.length) return [];
    var m = arrays.length;
    if (!weights || weights.length !== m) {
      // equal weights fallback
      weights = new Array(m);
      for (var wi = 0; wi < m; wi++) weights[wi] = 1 / m;
    }
    var n = arrays[0].length;
    var out = new Array(n).fill(0);
    for (var a = 0; a < m; a++){
      var w = weights[a];
      var arr = arrays[a];
      if (!arr || arr.length !== n) continue;
      for (var j = 0; j < n; j++) out[j] += w * arr[j];
    }
    var s = out.reduce((x,y)=>x+y,0) || 1;
    for (var k = 0; k < n; k++) out[k] /= s;
    return out;
  }

function __safeLog__(p){ return Math.log(Math.max(1e-12, Math.min(1, p))); }

// [CHG] Compute weighted multi-label log-loss for one draw
// Returns smaller = better loss; we'll convert to a "higher is better" score.
function __logLossForCase__(pMain, pExtra, truthMain, truthExtra, pickSize, extraCount, wMain, wExtra){
  var wm = Number((window.lottoConfig && window.lottoConfig.scoreWMain)  || 5);
  var we = Number((window.lottoConfig && window.lottoConfig.scoreWExtra) || 2);
  if (wMain != null)  wm = wMain;
  if (wExtra != null) we = wExtra;

  // Main balls
  var lossM = 0, denomM = Math.max(1, pickSize);
  for (var i = 0; i < truthMain.length; i++){
    var idx = truthMain[i] - 1;
    var p   = (pMain && pMain[idx] != null) ? pMain[idx] : 0;
    lossM  += -__safeLog__(p);
  }
  lossM /= denomM;

  // Extra balls (if any)
  var haveExtra = !!(truthExtra && truthExtra.length && pExtra && pExtra.length);
  var lossE = 0, denomE = Math.max(1, extraCount);
  if (haveExtra){
    for (var j = 0; j < truthExtra.length; j++){
      var eIdx = truthExtra[j] - 1;
      var pe   = (pExtra[eIdx] != null) ? pExtra[eIdx] : 0;
      lossE   += -__safeLog__(pe);
    }
    lossE /= denomE;
  }

  var totalW = wm + (haveExtra ? we : 0);
  if (!(totalW > 0)) totalW = 1;
  return (wm * lossM + (haveExtra ? we * lossE : 0)) / totalW;
}

function popularityProbs(allDraws, halfLife){
  var Nmain  = maxMain;
  var Nextra = hasExtra ? maxExtra : 0;

  var fM = new Float64Array(Nmain);
  var fE = hasExtra ? new Float64Array(Nextra) : null;

  var last = allDraws.length - 1;

  // Accumulate decayed frequencies
  for (var idx = 0; idx < allDraws.length; idx++) {
    var seq = allDraws[idx];
    var w   = decayWeight(last - idx, halfLife);

    var m = seq.slice(0, pickSize);
    for (var i = 0; i < m.length; i++) {
      var ii = m[i] - 1;
      if (ii >= 0 && ii < Nmain) fM[ii] += w;
    }

    if (hasExtra && fE) {
      var e = seq.slice(pickSize, pickSize + extraCount);
      for (var j = 0; j < e.length; j++) {
        var jj = e[j] - 1;
        if (jj >= 0 && jj < Nextra) fE[jj] += w;
      }
    }
  }

  // Normalize mains
  var sM = 0;
  for (var a = 0; a < Nmain; a++) sM += fM[a];
  if (!(sM > 0)) sM = 1;

  var pM = new Array(Nmain);
  for (var a2 = 0; a2 < Nmain; a2++) pM[a2] = fM[a2] / sM;

  // Normalize extras (if any)
  var pE = null;
  if (hasExtra && fE) {
    var sE = 0;
    for (var b = 0; b < Nextra; b++) sE += fE[b];
    if (!(sE > 0)) sE = 1;

    pE = new Array(Nextra);
    for (var b2 = 0; b2 < Nextra; b2++) pE[b2] = fE[b2] / sE;
  }

  return { main: pM, extra: pE };
}


  function mixRowStochastic(mats, weights){
    if (!Array.isArray(mats) || !mats.length || !Array.isArray(mats[0])) return [];
    var n = mats[0].length, out = new Array(n);
    for (var r = 0; r < n; r++){
      var row = new Float64Array(n);
      for (var m = 0; m < mats.length; m++){
        var w = (weights && weights[m] != null) ? Number(weights[m]) : 0;
        var R = mats[m][r] || [];
        for (var c = 0; c < n; c++) row[c] += w * (R[c] || 0);
      }
      var s = 0; for (var c2 = 0; c2 < n; c2++) s += row[c2];
      if (!(s > 0) || !isFinite(s)) {
        var u = n ? 1/n : 0; out[r] = Array.from({length:n}, _=>u);
      } else {
        var arr = new Array(n);
        for (var c3 = 0; c3 < n; c3++) arr[c3] = row[c3] / s;
        out[r] = arr;
      }
    }
    return out;
  }

  function buildOptimizedTransition(allDraws, K, decay) {
    // [CHG] Expand decay dynamic range so columns differ meaningfully.
// D small  -> long memory (large halfLife)
// D large  -> short memory (small halfLife)
var D = Math.max(0.001, Number(decay));
var halfLife = Math.max(5, Math.min(1000, Math.round(1 / D))); // allow HL up to 1000 draws


    var lag = buildLag1Transitions(allDraws, K, halfLife);
    var coo = buildCooccurrenceTransitions(allDraws, K, halfLife);
    var pop = popularityProbs(allDraws, halfLife);

    var pLag  = stationaryDistribution(lag.main);
    var pCoo  = stationaryDistribution(coo.main);
    var pPop  = pop.main;
    var N     = pLag.length;

    function toDiag(v) {
      var M = new Array(N);
      for (var r = 0; r < N; r++) {
        var row = new Array(N).fill(0);
        row[r] = v[r] || 0;
        M[r] = row;
      }
      return M;
    }

    var matsMain = [ lag.main, coo.main, toDiag(pPop) ];
    var weightsMain = (Array.isArray(window.__ENSEMBLE_WEIGHTS_MAIN__) && window.__ENSEMBLE_WEIGHTS_MAIN__.length === 3)
      ? window.__ENSEMBLE_WEIGHTS_MAIN__ : [0.40, 0.35, 0.25];

    var mixedMain = mixRowStochastic(matsMain, weightsMain);

    // Optional: expose blended extra vector for UI
    var extraEnsemble = null;
    if (hasExtra && (lag.extra || coo.extra || pop.extra)) {
      var pLagE = lag.extra ? stationaryDistribution(lag.extra) : null;
      var pCooE = coo.extra ? stationaryDistribution(coo.extra) : null;
      var pPopE = pop.extra || null;
      var parts = [], ws = (Array.isArray(window.__ENSEMBLE_WEIGHTS_EXTRA__) ? window.__ENSEMBLE_WEIGHTS_EXTRA__ : [0.40,0.35,0.25]);
      if (pLagE) { parts.push(pLagE); }
      if (pCooE) { parts.push(pCooE); }
      if (pPopE) { parts.push(pPopE); }
      if (parts.length) extraEnsemble = blendProbs(parts, ws.slice(0, parts.length));
    }
    if (extraEnsemble) window.__MCMC_EXTRA_ENS__ = extraEnsemble;

    return { main: mixedMain };
  }

  function createWorkerOrNull() { return null; } // main-thread fallback for now

  function runMCMCOnMainThread(cfg, onProgress, onDone){
    try {
      var trans = cfg && cfg.trans && cfg.trans.main ? cfg.trans.main : null;
      if (!trans || !trans.length || !trans[0] || !trans[0].length) throw new Error("Invalid transition matrix");
      var W  = Math.max(1, cfg.NUM_WALKS|0);
      var B  = Math.max(0, cfg.BURN_IN|0);
      var CL = Math.max(1, cfg.chainLen|0);
      var PC = (cfg.progressChunk && cfg.progressChunk > 0) ? cfg.progressChunk : Math.max(1, Math.floor(W / 100));
      var N  = trans.length;
      var visit = new Uint32Array(N);

      var rng = (function seedMulberry32(a){
        return function(){
          var t = a += 0x6D2B79F5;
          t = Math.imul(t ^ (t >>> 15), t | 1);
          t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
          return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
        };
      })((cfg.seed|0) || (Date.now()|0));

      function pickNext(state){
        var row = trans[state], r = rng(), acc = 0;
        for (var i = 0; i < row.length; i++){ acc += row[i]; if (r <= acc) return i; }
        return row.length - 1;
      }

      var w = 0;
      (function loopChunk(){
        var limit = Math.min(W, w + PC);
        for (; w < limit; w++){
          var s = Math.floor(rng() * N);
          for (var b = 0; b < B;  b++) s = pickNext(s);
          for (var t = 0; t < CL; t++) { s = pickNext(s); visit[s]++; }
        }
        onProgress && onProgress({ type: 'progress', completed: w });
        if (w < W) setTimeout(loopChunk, 0);
        else onDone && onDone({ type: 'result', visit: Array.from(visit) });
      })();
    } catch (err) {
      onDone && onDone({ type: 'error', message: String(err && err.message || err) });
    }
  }

  function spawnMCMC(cfg, onProgress, onDone) {
    var w = createWorkerOrNull();
    try { if (onProgress) onProgress({ type:'progress', completed:0 }); } catch(_e){}
    if (w) {
      var firstMsg = false, watchdogFired = false;
      var watchdog = setTimeout(function(){
        if (!firstMsg) {
          watchdogFired = true;
          try { w.terminate(); } catch(_e){}
          runMCMCOnMainThread(cfg, onProgress, onDone);
        }
      }, 1200);
      w.onerror = function(){ clearTimeout(watchdog); try{ w.terminate(); }catch(_e){}; runMCMCOnMainThread(cfg, onProgress, onDone); };
      w.onmessage = function (ev) {
        firstMsg = true; if (!ev || !ev.data) return;
        if (ev.data.type === 'progress') onProgress && onProgress(ev.data);
        else {
          clearTimeout(watchdog);
          if (!watchdogFired) { try{ w.terminate(); }catch(_e){} }
          onDone && onDone(ev.data);
        }
      };
      try { w.postMessage(cfg); } catch { clearTimeout(watchdog); runMCMCOnMainThread(cfg, onProgress, onDone); }
    } else {
      runMCMCOnMainThread(cfg, onProgress, onDone);
    }
  }

  function nextStepProbsFromLastDraw(allDraws, K, decay, pickSize, extraCount, maxMain, maxExtra, hasExtra){
    // [CHG] Same widened mapping as above to avoid collapse to a single halfLife.
var D = Math.max(0.001, Number(decay));
var halfLife = Math.max(5, Math.min(1000, Math.round(1 / D))); // allow HL up to 1000 draws


    var last = allDraws[allDraws.length - 1] || [];
    var lastMainIdx = last.slice(0, pickSize).map(function(n){ return n - 1; }).filter(function(i){ return i >= 0 && i < maxMain; });
    var lastExtraIdx = (hasExtra && extraCount > 0)
      ? last.slice(pickSize, pickSize + extraCount).map(function(n){ return n - 1; }).filter(function(i){ return i >= 0 && i < maxExtra; })
      : [];

    var lag = buildLag1Transitions(allDraws, K, halfLife);
    var coo = buildCooccurrenceTransitions(allDraws, K, halfLife);
    var pop = popularityProbs(allDraws, halfLife);

    var pLagMain = averageRows(lag.main, lastMainIdx, maxMain) || stationaryDistribution(lag.main);
    var pCooMain = averageRows(coo.main, lastMainIdx, maxMain) || stationaryDistribution(coo.main);
    var wMain = window.__ENSEMBLE_WEIGHTS_MAIN__ || [0.40,0.35,0.25];
    var pMain = blendProbs([pLagMain, pCooMain, pop.main], wMain);

    var pExtra = null;
    if (hasExtra) {
      var wE = window.__ENSEMBLE_WEIGHTS_EXTRA__ || [0.40,0.35,0.25];
      var pLagE = lag.extra ? (averageRows(lag.extra, lastExtraIdx, maxExtra) || stationaryDistribution(lag.extra)) : null;
      var pCooE = coo.extra ? (averageRows(coo.extra, lastExtraIdx, maxExtra) || stationaryDistribution(coo.extra)) : null;
      var parts = [], ws = [];
      if (pLagE)  { parts.push(pLagE);  ws.push(wE[0]); }
      if (pCooE)  { parts.push(pCooE);  ws.push(wE[1]); }
      if (pop.extra) { parts.push(pop.extra); ws.push(wE[2]); }
      pExtra = parts.length ? blendProbs(parts, ws) : null;
    }
    return { main: pMain, extra: pExtra };
  }

function buildBacktestRows(topN, win){
  var rows = [];
  var draws = Array.isArray(window.lottoDraws) ? window.lottoDraws : [];
  var dates = Array.isArray(window.lottoDrawDates) ? window.lottoDrawDates : [];
  var cfg   = window.lottoConfig || {};
  var pick  = Number(cfg.pickSize)||0;
  var hasE  = !!cfg.hasExtra;
  var eCnt  = Number(cfg.extraCount)||0;
  var maxM  = Array.isArray(cfg.rangeMain)  ? Number(cfg.rangeMain[1])  : 0;
  var maxE  = Array.isArray(cfg.rangeExtra) ? Number(cfg.rangeExtra[1]) : 0;

  if (draws.length < 5 || !pick || !maxM) return rows;

  // Use the current UI settings for K & Decay (the same logic you predict with)
  var K  = Number(document.getElementById("inputSmooth")?.value) || 1;
  var D  = Number(document.getElementById("inputDecay")?.value)  || 0.02;

// Use the last N draws as chosen by the user (fallback to 200 if unset/invalid)
var countEl = document.getElementById('btCount');
var wantN   = parseInt(countEl && countEl.value, 10);
if (!Number.isFinite(wantN) || wantN < 20) wantN = 200;

// We evaluate t in [startIdx .. draws.length-2] (since truth = draws[t+1])
var maxEval  = Math.max(0, (draws.length - 1));   // last usable t
var startIdx = Math.max(4, maxEval - wantN);      // clamp with small warmup

for (var t = startIdx; t + 1 < draws.length; t++) {
  var startHist = (win && win > 0) ? Math.max(0, t - win + 1) : 0;
  var hist = draws.slice(startHist, t + 1);
  var probs = nextStepProbsFromLastDraw(hist, K, D, pick, eCnt, maxM, maxE, hasE);

    // Top-N main list
var mainTop = probs.main
  .map(function(p,i){ return {n:i+1, p:p}; })
  .sort(function(a,b){ return b.p - a.p; })
  .slice(0, Math.min(20, maxM)); // Top-20 mains

// For extras, show top eCnt (or up to 5 if you prefer)
var extraTop = [];
if (hasE && Array.isArray(probs.extra)) {
  extraTop = probs.extra
    .map(function(p,i){ return {n:i+1, p:p}; })
    .sort(function(a,b){ return b.p - a.p; })
    .slice(0, EXTRA_SHOW);
}

    var truth = draws[t+1] || [];
    var truthMain  = truth.slice(0, pick);
    var truthExtra = truth.slice(pick, pick + Math.max(0, eCnt));
    var hits = 0;
    var hitSet = {};
    for (var i=0;i<truthMain.length;i++){ hitSet[ truthMain[i] ] = 1; }
    mainTop.forEach(function(x){ if(hitSet[x.n]) hits++; });

    rows.push({
      idx: t+1,
      date: dates[t+1] || '',
      mainTop: mainTop.map(function(x){ return x.n; }),
      truthMain: truthMain.slice(),
      extraTop: extraTop.map(function(x){ return x.n; }),
      truthExtra: truthExtra.slice(),
      hitCount: hits
    });
  }
  return rows.reverse(); // newest first like your screenshot
}

// Build rows using the SAME MCMC engine/settings as "Run Prediction"
async function buildBacktestRowsMCMC(topN, win, onProgress){
  const draws = Array.isArray(window.lottoDraws) ? window.lottoDraws : [];
  const dates = Array.isArray(window.lottoDrawDates) ? window.lottoDrawDates : [];
  const cfg   = window.lottoConfig || {};
  const pick  = Number(cfg.pickSize)||0;
  const hasE  = !!cfg.hasExtra;
  const eCnt  = Number(cfg.extraCount)||0;
  const maxM  = Array.isArray(cfg.rangeMain)  ? Number(cfg.rangeMain[1])  : 0;
  const maxE  = Array.isArray(cfg.rangeExtra) ? Number(cfg.rangeExtra[1]) : 0;

  if (draws.length < 5 || !pick || !maxM) return [];

  // Use shared settings helper (exported on window by Backtest wiring)
  const defaults = { W:1500, B:50, K:1, D:0.02, CL:1500 };
  const cfgSettings = (window.getCurrentSettings ? window.getCurrentSettings() : defaults);
  const {W,B,K,D,CL} = cfgSettings;

  const rows = [];
// Use the last N draws chosen by the user (fallback 200 if unset/invalid)
var countEl = document.getElementById('btCount');
var wantN   = parseInt(countEl && countEl.value, 10);
if (!Number.isFinite(wantN) || wantN < 20) wantN = 200;

const maxEval  = Math.max(0, (draws.length - 1)); // last usable t (we need t+1 for truth)
const startIdx = Math.max(4, maxEval - wantN);    // clamp with small warmup
const total    = Math.max(0, (draws.length - 1) - startIdx);

  let done = 0, cancelled = false;

  // hook Cancel button
  const cancelBtn = document.getElementById('btnCancelBacktest');
  if (cancelBtn) {
    const handler = () => { cancelled = true; };
    cancelBtn.__btCancelHandler && cancelBtn.removeEventListener('click', cancelBtn.__btCancelHandler);
    cancelBtn.__btCancelHandler = handler;
    cancelBtn.addEventListener('click', handler);
  }

  for (let t = startIdx; t + 1 < draws.length; t++){
    if (cancelled) break;

    const startHist = (win && win > 0) ? Math.max(0, t - win + 1) : 0;
    const hist = draws.slice(startHist, t + 1);

    // Same blended transition as Prediction
    const transObj = buildOptimizedTransition(hist, K, D);

    // Run SAME MCMC as Prediction for this step
    const pMain = await new Promise((resolve) => {
      spawnMCMC({
        trans: transObj,
        NUM_WALKS: W,
        BURN_IN: B,
        chainLen: CL,
        progressChunk: Infinity,                 // one result per historical step
        seed: (Math.floor(Date.now()%1e9) ^ t)  // vary seed per step
      }, null, (ev) => {
        if (!ev) return;
        if (ev.type === 'result') {
          const visit = Array.isArray(ev.visit) ? ev.visit.slice(0, maxM) : [];
          const s = visit.reduce((a,b)=>a+b,0) || 1;
          resolve(visit.map(v => v / s));
        }
      });
    });

    // Extras vector from buildOptimizedTransition()
    const pExtra = (hasE && window.__MCMC_EXTRA_ENS__ && Array.isArray(window.__MCMC_EXTRA_ENS__))
      ? window.__MCMC_EXTRA_ENS__.slice(0, maxE) : null;

    // Top lists
    const mainTop = pMain
      .map((p,i)=>({n:i+1,p}))
      .sort((a,b)=>b.p-a.p)
      .slice(0, Math.max(topN, 1))
      .map(x=>x.n);

    const extraTop = (hasE && pExtra)
      ? pExtra.map((p,i)=>({n:i+1,p}))
              .sort((a,b)=>b.p-a.p)
              .slice(0, EXTRA_SHOW)
              .map(x=>x.n)
      : [];

    // Truth
    const truth     = draws[t+1] || [];
    const truthMain = truth.slice(0, pick);
    const truthExtra= truth.slice(pick, pick + Math.max(0, eCnt));

    // Hits
    const hitSet = Object.create(null);
    truthMain.forEach(n => { hitSet[n] = 1; });
    const hitCount = mainTop.reduce((acc,n)=>acc + (hitSet[n]?1:0), 0);

    rows.push({
      idx: t+1,
      date: dates[t+1] || '',
      mainTop,
      truthMain,
      extraTop,
      truthExtra,
      hitCount
    });

    done++;
    onProgress && onProgress(done, total);
    await new Promise(r => setTimeout(r, 0)); // yield to UI
  }

  return rows.reverse();
}


// Decode Sorcerer-style [[...]] into HTML and whitelist common tags used in BT table.
function __btDecode__(html){
  if (!html || html.indexOf('[[') === -1) return html;
  var out = html.replace(/\[\[/g, '<').replace(/]]/g, '>');
  // whitelist: span, div, br, table, thead, tbody, tr, th, td, ul, li, em, strong, code, small, b, i
  out = out.replace(/<(?!\/?(?:span|div|br|table|thead|tbody|tr|th|td|ul|li|em|strong|code|small|b|i)\b)[^>]*>/gi,
    function(m){ return m.replace(/</g,'&lt;').replace(/>/g,'&gt;'); });
  return out;
}

function downsampleEdgeMid(arr, target){
  if (!Array.isArray(arr) || arr.length <= target) return arr || [];
  var out = [];
  for (var i = 0; i < target; i++){
    var idx = Math.round(i * (arr.length - 1) / (target - 1));
    out.push(arr[idx]);
  }
  return out;
}
// [CHG] Log-spaced downsample with guards (keeps endpoints, avoids 0/NaN/dupes)
function logDownsample(minV, maxV, target){
  minV   = Math.max(1, Math.floor(Number(minV) || 0)); // walks/budgets must be >=1
  maxV   = Math.max(minV + 1, Math.floor(Number(maxV) || 0));
  target = Math.max(2, Math.floor(Number(target) || 2)); // need at least 2 points

  var out = [];
  var r = Math.pow(maxV / minV, 1 / (target - 1));
  var v = minV;
  for (var i = 0; i < target; i++){
    var w = Math.round(v);
    if (w >= minV && w <= maxV) out.push(w);
    v *= r;
  }
  // ensure endpoints are present
  out.push(minV, maxV);

  // dedupe + sort deterministically
  out.sort(function(a,b){ return a - b; });
  var uniq = [], last = null;
  for (var j = 0; j < out.length; j++){
    var k = out[j];
    if (k !== last) { uniq.push(k); last = k; }
  }
  return uniq;
}

}

function scoreParamsLeakFree(W, B, CL, K, D) {
  if (!Array.isArray(draws) || draws.length < 3) return 0;

  var swMain  = Number((window.lottoConfig && window.lottoConfig.scoreWMain)  || 5);
  var swExtra = Number((window.lottoConfig && window.lottoConfig.scoreWExtra) || 2);

  function __safeLog__(p){ if (!(p > 0)) p = 1e-12; if (p > 1) p = 1; return Math.log(p); }
  function __logLossForCase__(pMain, pExtra, truthMain, truthExtra){
    var lossM = 0, denomM = Math.max(1, pickSize);
    for (var i = 0; i < truthMain.length; i++){
      var idx = truthMain[i] - 1;
      var pm  = (pMain && pMain[idx] != null) ? pMain[idx] : 0;
      lossM  += -__safeLog__(pm);
    }
    lossM /= denomM;

    var haveExtra = !!(hasExtra && extraCount > 0 && pExtra && truthExtra && truthExtra.length);
    var lossE = 0, denomE = Math.max(1, extraCount);
    if (haveExtra){
      for (var j = 0; j < truthExtra.length; j++){
        var eIdx = truthExtra[j] - 1;
        var pe   = (pExtra[eIdx] != null) ? pExtra[eIdx] : 0;
        lossE   += -__safeLog__(pe);
      }
      lossE /= denomE;
    }
    var totalW = swMain + (haveExtra ? swExtra : 0);
    if (!(totalW > 0)) totalW = 1;
    return (swMain * lossM + (haveExtra ? swExtra * lossE : 0)) / totalW;
  }

  // geometric/stratified checkpoints with slight recency emphasis
  var start = Math.max(10, Math.floor(draws.length * 0.60)); // evaluate on last 40%
  var endEx = draws.length - 1;                              // we use truth = draws[t+1]
  if (endEx - start <= 0) return 0;

  var casesMin = (draws.length < 800 ? 30 : 40);
  var casesMax = (draws.length > 4000 ? 100 : 80);

  // Requires buildCheckpoints(start, endEx, casesMin, casesMax) in Helpers section
  var checkpoints = buildCheckpoints(start, endEx, casesMin, casesMax);
  if (!checkpoints.length) return 0;

  var totalLoss = 0, cases = 0;
  for (var c = 0; c < checkpoints.length; c++){
    var t          = checkpoints[c];
    var hist       = draws.slice(0, t + 1);
    var truth      = draws[t + 1];
    var truthMain  = truth.slice(0, pickSize);
    var truthExtra = truth.slice(pickSize, pickSize + Math.max(0, extraCount));

    var ns = nextStepProbsFromLastDraw(hist, K, D, pickSize, extraCount, maxMain, maxExtra, hasExtra);
    var lossThis = __logLossForCase__(ns.main, ns.extra, truthMain, truthExtra);
    if (isFinite(lossThis)) { totalLoss += lossThis; cases++; }
  }

  var meanLoss = cases ? (totalLoss / cases) : Infinity;
  return (isFinite(meanLoss) ? Math.round(1e6 / (1 + meanLoss)) : 0); // higher = better
}
